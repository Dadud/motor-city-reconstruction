option(MCO_BUILD_PLATFORM "Build the SDL/OpenGL platform probe and asset garage" OFF)
if(MCO_BUILD_PLATFORM)
  set(SDL_TESTS OFF CACHE BOOL "" FORCE)
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
  FetchContent_Declare(sdl_source
    URL https://www.libsdl.org/release/SDL3-3.2.28.tar.gz
    URL_HASH SHA256=1330671214d146f8aeb1ed399fc3e081873cdb38b5189d1f8bb6ab15bbc04211
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(sdl_source)
  add_executable(mco_platform_probe native/src/platform_main.cpp)
  target_link_libraries(mco_platform_probe PRIVATE SDL3::SDL3)
endif()

# This target contains recovered calculations, not prototype vehicle handling.
file(GLOB mco_recovered_sources CONFIGURE_DEPENDS native/src/*.cpp)
list(FILTER mco_recovered_sources EXCLUDE REGEX "/(mod_script|mod_script_main|script_runtime|garage_main|platform_main)\\.cpp$")
add_library(mco_core ${mco_recovered_sources})
target_include_directories(mco_core PUBLIC native/include)
if(MCO_BUILD_PLATFORM)
  add_executable(mco_native_garage native/src/garage_main.cpp)
  target_link_libraries(mco_native_garage PRIVATE mco_core SDL3::SDL3)
endif()
if(MSVC)
  target_compile_options(mco_core PRIVATE /W4 /fp:strict)
else()
  target_compile_options(mco_core PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off -fno-fast-math)
endif()

option(MCO_BUILD_SCRIPTING "Build the bounded Lua research runtime" ON)
if(MCO_BUILD_SCRIPTING)
  FetchContent_Declare(lua_source
    URL https://www.lua.org/ftp/lua-5.4.8.tar.gz
    URL_HASH SHA256=4f18ddae154e793e46eeab727c59ef1c0c0c2b744e7b94219710d76f530629ae
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(lua_source)
  file(GLOB lua_sources CONFIGURE_DEPENDS "${lua_source_SOURCE_DIR}/src/*.c")
  list(FILTER lua_sources EXCLUDE REGEX "/(lua|luac)\\.c$")
  add_library(mco_lua STATIC ${lua_sources})
  target_include_directories(mco_lua PUBLIC "${lua_source_SOURCE_DIR}/src")
  if(UNIX)
    target_link_libraries(mco_lua PUBLIC m)
  endif()
  add_library(mco_scripting native/src/script_runtime.cpp native/src/mod_script.cpp)
  target_include_directories(mco_scripting PUBLIC native/include)
  target_link_libraries(mco_scripting PRIVATE mco_lua)
  add_executable(mco_mod_script native/src/mod_script_main.cpp)
  target_link_libraries(mco_mod_script PRIVATE mco_scripting)
endif()
